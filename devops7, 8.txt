# ─────────────────────────────────────────────────
#  Stage 1 – Build: Copy static website assets
# ─────────────────────────────────────────────────
FROM nginx:alpine
# Maintainer label
LABEL maintainer="girlinfog" \
      project="program2-master" \
      description="Tripillar Solutions static website served via Nginx"
# Remove default Nginx welcome page
RUN rm -rf /usr/share/nginx/html/*
# Copy static website files into Nginx's serving directory
COPY src/main/resources/index.html    /usr/share/nginx/html/index.html
COPY src/main/resources/styles.css    /usr/share/nginx/html/styles.css
COPY src/main/resources/logo-bit.jpg.jpeg   /usr/share/nginx/html/logo-bit.jpg.jpeg
# Expose port 80 (Nginx default)
EXPOSE 80
# Start Nginx in the foreground (required for Docker)
CMD ["nginx", "-g", "daemon off;"]
